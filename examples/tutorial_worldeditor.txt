// A small computer player in Bullfrog syntax.
{
  IF ( INT_GAME_TURN == 0 )
  {
    // Things the tribe does on its own
    DO STATE_CONSTRUCT_BUILDING ON
    DO STATE_FETCH_WOOD ON
    DO STATE_TRAIN_PEOPLE ON
    SET INT_ATTR_EXPANSION 12
    SET INT_ATTR_HOUSE_PERCENTAGE 60
    SET $USER_ATTACKS 0
    SET $USER_RESULT 0
    DO SET_ATTACK_VARIABLE $USER_RESULT
  }
  ENDIF

  // Every 512 turns: attack blue if there are enough warriors
  EVERY 512 64
  {
    IF ( INT_M_PERSON_WARRIOR > 5 && INT_RANDOM_100 < 70 )
    {
      DO ATTACK BLUE 6 ATTACK_BUILDING INT_DRUM_TOWER 999 INT_BLAST INT_NO_SPECIFIC_SPELL INT_NO_SPECIFIC_SPELL ATTACK_NORMAL 0 -1 -1 -1
      INCREMENT $USER_ATTACKS 1
    }
    ELSE
    {
      DO TRAIN_PEOPLE_NOW 2 INT_WARRIOR  // not enough yet
    }
    ENDIF
  }
}
SCRIPT_END
